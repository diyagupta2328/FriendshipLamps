#include "ti_msp_dl_config.h"
#include <string.h>
#include <stdbool.h>
#include <stdint.h>

#define RX_BUF_SIZE 256

volatile char     rxBuffer[RX_BUF_SIZE];
volatile uint32_t rxIndex    = 0;
volatile bool     lineReady  = false;
volatile int      atResult   = -1;
volatile int      uartResult = -1;

void uart_send_string(const char *str)
{
    while (*str)
    {
        DL_UART_Main_transmitDataBlocking(UART_0_INST, (uint8_t)*str);
        str++;
    }
}

void delay_ms(uint32_t ms)
{
    delay_cycles(32000UL * ms);
}

void clear_rx_buffer(void)
{
    rxIndex   = 0;
    lineReady = false;
    memset((void *)rxBuffer, 0, sizeof(rxBuffer));
}

/*
 * Search for "OK" anywhere in rxBuffer.
 * The ESP32 may echo the command first, then send OK.
 * So we scan the whole buffer rather than just the end.
 */
int bufferContainsOK(void)
{
    for (uint32_t i = 0; i + 1 < rxIndex; i++)
    {
        if (rxBuffer[i] == 'O' && rxBuffer[i+1] == 'K')
            return 1;
    }
    return 0;
}

int main(void)
{
    SYSCFG_DL_init();
    NVIC_ClearPendingIRQ(UART_0_INST_INT_IRQN);
    NVIC_EnableIRQ(UART_0_INST_INT_IRQN);

    /* Wait for ESP32 to fully boot */
    delay_ms(3000);

    /* ── Test 1: Basic AT ── */
    clear_rx_buffer();
    uart_send_string("AT\r\n");

    /* Wait 2 seconds for any response including echo + OK */
    delay_ms(2000);
    atResult = bufferContainsOK() ? 1 : 0;

    /* Save a snapshot of what AT returned */
    char atSnapshot[32];
    memcpy(atSnapshot, (void*)rxBuffer, 31);
    atSnapshot[31] = '\0';

    delay_ms(500);

    /* ── Test 2: Change UART permanently to GPIO1/GPIO3 ── */
    clear_rx_buffer();
    uart_send_string("AT+UART_DEF=115200,8,1,0,0\r\n");

    /* Wait 3 seconds for response */
    delay_ms(3000);
    uartResult = bufferContainsOK() ? 1 : 0;

    /*
     * SET BREAKPOINT ON delay_ms BELOW.
     * Watch window:
     *   atResult   = 1 means AT worked
     *   uartResult = 1 means UART change accepted
     *   rxBuffer   = full response from ESP32
     *   atSnapshot = what the AT command returned
     */
    delay_ms(10000);  /* ← BREAKPOINT HERE */

    while (1) { delay_ms(1000); }
}

void UART_0_INST_IRQHandler(void)
{
    switch (DL_UART_Main_getPendingInterrupt(UART_0_INST))
    {
        case DL_UART_MAIN_IIDX_RX:
        {
            char c = (char)DL_UART_Main_receiveData(UART_0_INST);
            if (rxIndex < RX_BUF_SIZE - 1)
            {
                rxBuffer[rxIndex] = c;
                rxIndex++;
            }
            break;
        }
        default:
            break;
    }
}
